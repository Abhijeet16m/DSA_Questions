#include<bits/stdc++.h>
using namespace std;

int solution(vector<int> &v){
    int i = 0, j = 1;
    int res = 1;
    while(j < v.size()){
        if(v[j] < v[j-1]) i = j;
        res = max(res, j-i+1);
        j++;
    }
    return res;
}

int main(){
    int n; cin>>n;
    vector<int> v(n);
    for(int i = 0; i < n; i++) cin>>v[i];
    cout<<solution(v)<<endl;
    return 0;
}
