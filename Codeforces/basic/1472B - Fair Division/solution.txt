#include<bits/stdc++.h>
using namespace std;

bool solution(vector<int> &v){
    int one = 0, two = 0;
    for(int i: v){
        if(i == 1) one++;
        if(i == 2) two++;
    }
    if(two%2 && one < 2) return false;
    return one%2 == 0;
}

int main(){
    int t;
    cin>>t;
    while(t--){
        int n; cin>>n;
        vector<int> v(n);
        for(int i = 0; i < n; i++) cin>>v[i];
        if(solution(v)) cout<<"yes";
        else cout<<"no";
        cout<<endl;
    }
    return 0;
}
