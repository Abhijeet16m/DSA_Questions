#include<bits/stdc++.h>
using namespace std;

int helper(vector<int> &v, int target){
    int left = 0, right = v.size() - 1;
    int upperBound = -1;
    int flag = 0;
    while (left <= right)
    {
        int mid = left + (right - left)/2;
        if(v[mid] == target){
            flag = 1;
        }
        if(v[mid] > target){
            right = mid - 1;
        }
        else{
            upperBound = mid;
            left = mid+1;
        }
    }
    return upperBound+1;

    
}

int main(){
    int n; cin>>n;
    vector<int> v(n);
    for(int i = 0; i < n; i++) cin>>v[i];
    sort(v.begin(), v.end());
    int q; cin>>q;
    for(int i = 0; i < q; i++){
        int m; cin>>m;
        cout<<helper(v, m)<<endl;
    }
    return 0;
}
