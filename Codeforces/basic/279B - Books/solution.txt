#include<bits/stdc++.h>
using namespace std;

int solution(vector<int> &v, int n, int t){
    int res = 0;
    int i = 0, j = 0;
    int currSum = 0;
    while(j < v.size()){
        currSum += v[j];
        while(currSum > t){
            currSum -= v[i];
            i++;    
        }
        res = max(res, j-i+1);
        j++;
    }
    return res;
}

int main(){
    int n, t;
    cin>>n>>t;
    vector<int> v(n);
    for(int i = 0;  i< n; i++) cin>>v[i];
    cout<<solution(v, n, t);
    return 0;
}
