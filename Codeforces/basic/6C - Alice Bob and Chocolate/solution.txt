#include<bits/stdc++.h>
using namespace std;

pair<int, int> solution(vector<int> &v){
    int totalSum = 0;
    for(int i: v) totalSum += i;
    int temp = 0;
    int alice = 0, bob = 0;
    for(int i: v){
        totalSum -= i;
        if(totalSum < temp) bob++;
        else alice++;
        temp += i;
    }
    return {alice, bob};
    
}

int main(){
    int n;
    cin>>n;
    vector<int> v(n);
    for(int i = 0; i < n; i++){
        cin>>v[i];
    }
    pair<int, int> ans = solution(v);
    cout<<ans.first<<" "<<ans.second<<endl;
    return 0;
}
