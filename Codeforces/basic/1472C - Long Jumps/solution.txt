#include<bits/stdc++.h>
using namespace std;

class Solution{
    private:
        vector<int> dp;
    public:
        int helper(vector<int> &v, int i, int n){
            if(i >= n) return 0;
            if(dp[i] != -1) return dp[i];
            return dp[i] = v[i]+helper(v, i+v[i], n);
        }
        int longJumps(vector<int> &v, int n){
            dp.resize(n, -1);
            int res = INT_MIN; // since n is given always greater than or equal to 1; I will never return INT_MIN;
            for(int i = 0; i < n; i++){
                int temp = helper(v, i, n);
                res = max(res, temp);
            }
            return res;
        }
};

int main(){
    int t; cin>>t;
    while (t--)
    {
        int n; cin>>n;
        vector<int> v(n);
        Solution s = Solution();
        for(int i = 0; i < n; i++) cin>>v[i];
        cout<<s.longJumps(v, n)<<endl;
    }
    return 0;
}
