#include<bits/stdc++.h>
using namespace std;

int helper(vector<int> &v, int target){
    int res = -1;
    int left = 0, right = v.size()-1;
    while (left <= right)
    {
        int mid = left + (right - left)/2;
        if(v[mid] == target) return mid+1;
        else if(v[mid] > target){
            res = mid;
            right = mid - 1;
        }
        else{
            left = mid + 1;
        }
    }
    return res+1;
}
int main(){
    int n; cin>>n;
    vector<int> v(n);
    for(int i = 0; i < n; i++) cin>>v[i];
    int q; cin>>q;
    vector<int> t(q);
    for(int i = 0; i < q; i++) cin>>t[i];
    vector<int> prefix;
    int currSum = 0;
    for(int i: v){
        currSum += i;
        prefix.push_back(currSum);
    }
    for(int i: t){
        cout<<helper(prefix, i)<<endl;
    }
    return 0;
}
